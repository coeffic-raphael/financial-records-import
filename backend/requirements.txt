# Runtime dependencies, pinned.
#
# REQUIRES PYTHON 3.11 OR NEWER. The code uses StrEnum and datetime.UTC, both
# introduced in 3.11; on 3.10 the very first import fails.
#
# Direct dependencies only. Transitive ones (starlette, anyio, Mako, ...) are
# left to resolve. See "Production improvements" in the README for what a full
# lockfile would buy and why it is not used here.

fastapi==0.141.1
# The [email] extra pulls in email-validator, which EmailStr needs at import
# time. Without it the application fails to start, not at first use.
pydantic[email]==2.13.4
pydantic-settings==2.15.0
sqlalchemy==2.0.52
alembic==1.19.1

# Required by FastAPI to parse multipart uploads. It appears in no import
# statement in this codebase, so it looks removable -- it is not: dropping it
# breaks CSV upload at runtime.
python-multipart==0.0.32

# ASGI server. Plain rather than uvicorn[standard]: the extras compile C
# extensions, which makes a reviewer's install slower and more fragile for no
# gain that matters here.
uvicorn==0.52.3

# Authentication.
# argon2-cffi rather than bcrypt: Argon2id is the current password-hashing
# recommendation, and bcrypt silently truncates beyond 72 bytes.
argon2-cffi==25.1.0
pyjwt==2.13.0

# AI providers. Both are imported lazily, so the application starts and serves
# every non-extraction endpoint even with no credentials configured.
google-genai==2.18.1
openai==3.1.0
